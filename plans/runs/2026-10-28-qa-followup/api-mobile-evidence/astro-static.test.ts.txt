import { afterAll, beforeAll, describe, expect, it, vi } from 'vitest';
import { Effect, Fiber, Layer } from 'effect';
import { HttpRouter, HttpServer } from 'effect/http';
import { NodeHttpPlatform, NodeHttpServer, NodeServices } from '@effect/platform-node';
import { createServer } from 'node:http';
import { resolve } from 'node:path';
import { boundary, records, staticApp } from '../src/app.ts';
const root = resolve('apps/web/dist');
const layer = staticApp(root).pipe(Layer.provide([NodeServices.layer, NodeHttpPlatform.layer]));
const runtime = HttpRouter.toWebHandler(layer.pipe(Layer.provide(HttpServer.layerServices)), { disableLogger: true, middleware: boundary });
const server = createServer();
const running = Effect.runFork(Layer.launch(HttpRouter.serve(layer, { disableLogger: true, disableListenLog: true, middleware: boundary }).pipe(Layer.provide(NodeHttpServer.layer(() => server, { port: 0, host: '127.0.0.1' })))));
beforeAll(async () => { await vi.waitFor(() => { if (!server.listening) throw Error('not ready'); }); });
afterAll(async () => { await runtime.dispose(); await Effect.runPromise(Fiber.interrupt(running)); });
for (const transport of ['web', 'node'] as const) describe(transport, () => {
  const request = (path: string) => { const url = transport === 'web' ? 'http://local' : `http://127.0.0.1:${(server.address() as {port:number}).port}`; return transport === 'web' ? runtime.handler(new Request(url + path)) : fetch(url + path); };
  for (const [path, status, text] of [['/', 200, 'Built QA page'], ['/docs/', 200, 'Built directory page'], ['/missing', 404, 'Built site missing page']] as const) it(path, async () => {
    records.length = 0; const res = await request(path); expect(res.status).toBe(status); expect(await res.text()).toContain(text); expect(res.headers.get('x-request-id')).toBeTruthy(); expect(records.filter(r => r.msg === 'request')).toHaveLength(1);
  });
  it('keeps an API miss in the API envelope', async () => { records.length = 0; const res = await request('/v1/missing'); expect(res.status).toBe(404); expect((await res.json()).error.tag).toBe('NotFound'); expect(records.filter(r => r.msg === 'request')).toHaveLength(1); });
});
