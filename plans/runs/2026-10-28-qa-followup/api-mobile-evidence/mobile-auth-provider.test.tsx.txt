import React from 'react';
import { Text } from 'react-native';
import { render, waitFor } from '@testing-library/react-native';
import { QueryClient } from '@tanstack/react-query';
import { RootProviders } from '../src/app/_layout';
import { useHealth } from '../src/lib/queries/useHealth';
import { setAuth, providerMounts } from './clerk-mock';
jest.mock('@clerk/expo', () => require('./clerk-mock'));
function Screen({ enabled }: {enabled: boolean}) { const result = useHealth(); return <Text>{result.isError ? 'error' : result.data?.status ?? 'loading'}</Text>; }
const make = (enabled: boolean, client: QueryClient) => <RootProviders enabled={enabled} client={client}><Screen enabled={enabled} /></RootProviders>;
test('anonymous, token refresh, token failure and sign-out use the real hook and client', async () => {
  const calls: Array<string | null> = [];
  const logs: string[] = [];
  jest.spyOn(console, 'log').mockImplementation((...args) => { logs.push(JSON.stringify(args)); });
  jest.spyOn(console, 'error').mockImplementation((...args) => { logs.push(JSON.stringify(args)); });
  jest.spyOn(globalThis, 'fetch').mockImplementation(async (_url, init) => { calls.push(new Headers(init?.headers).get('authorization')); return new Response('{"status":"ok"}', { status: 200, headers: {'content-type':'application/json'} }); });
  const client = new QueryClient({ defaultOptions: { queries: { retry: false, gcTime: Infinity } } });
  setAuth({ isSignedIn: false, getToken: async () => null });
  const view = await render(make(false, client));
  await waitFor(() => expect(calls).toEqual([null]));
  expect(providerMounts).toBe(0);
  await view.unmount(); client.clear(); calls.length = 0;
  const signedOut = await render(make(true, client));
  await waitFor(() => expect(calls).toEqual([null]));
  await signedOut.unmount(); client.clear(); calls.length = 0;
  let token = 'first'; setAuth({ isSignedIn: true, getToken: async () => token });
  const signed = await render(make(true, client));
  await waitFor(() => expect(calls).toEqual(['Bearer first']));
  token = 'second'; await client.invalidateQueries({ queryKey: ['health'] });
  await waitFor(() => expect(calls).toEqual(['Bearer first', 'Bearer second']));
  setAuth({ isSignedIn: true, getToken: async () => { throw Error('token failure'); } });
  await signed.rerender(make(true, client)); await client.invalidateQueries({ queryKey: ['health'] });
  await waitFor(() => expect(signed.getByText('error')).toBeTruthy());
  expect(calls).toHaveLength(2);
  setAuth({ isSignedIn: false, getToken: async () => null });
  await signed.rerender(make(true, client));
  await waitFor(() => expect(client.getQueryData(['health'])).toBeUndefined());
  await signed.unmount();
  const anonymousAgain = await render(make(true, client));
  await waitFor(() => expect(calls.at(-1)).toBeNull());
  expect(logs.join()).not.toContain('first'); expect(logs.join()).not.toContain('second');
  await anonymousAgain.unmount(); jest.restoreAllMocks();
});
