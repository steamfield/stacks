name: CI
on: [push, pull_request, workflow_dispatch]
permissions: { contents: read }
jobs:
  changes:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    outputs:
      api: '${{ steps.filter.outputs.api }}'
      mobile: '${{ steps.filter.outputs.mobile }}'
    steps:
      - uses: actions/checkout@v6
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          filters: |
            api:
              - 'apps/api/**'
              - 'packages/schemas/**'
              - 'bun.lock'
              - 'package.json'
              - 'bunfig.toml'
              - '.node-version'
              - 'drizzle.config.ts'
              - '.github/workflows/**'
            mobile:
              - 'apps/mobile/**'
              - 'apps/api/**'
              - 'packages/schemas/**'
              - 'bun.lock'
              - 'package.json'
              - 'bunfig.toml'
              - '.node-version'
              - 'drizzle.config.ts'
              - '.github/workflows/**'
  lint:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v6
      - uses: oven-sh/setup-bun@v2
        with: { bun-version-file: package.json }
      - run: bun install --frozen-lockfile
      - run: bun run lint
      - run: bun run test:ci
  typecheck:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v6
      - uses: oven-sh/setup-bun@v2
        with: { bun-version-file: package.json }
      - run: bun install --frozen-lockfile
      - run: bun run typecheck
  api:
    needs: changes
    if: "${{ needs.changes.outputs.api == 'true' }}"
    runs-on: ubuntu-latest
    timeout-minutes: 10
    services:
      postgres:
        image: postgres:18-alpine
        env: { POSTGRES_USER: qa, POSTGRES_PASSWORD: qa, POSTGRES_DB: qa }
        ports: ['5432:5432']
        options: '--health-cmd "pg_isready -U qa" --health-interval 5s --health-timeout 5s --health-retries 10'
    env: { DATABASE_URL: 'postgres://qa:qa@localhost:5432/qa', QA_RUN_ID: ci }
    steps:
      - uses: actions/checkout@v6
      - uses: oven-sh/setup-bun@v2
        with: { bun-version-file: package.json }
      - run: bun install --frozen-lockfile
      - run: bun run db:check
      - run: bun run test:api
  mobile:
    needs: changes
    if: "${{ needs.changes.outputs.mobile == 'true' }}"
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v6
      - uses: oven-sh/setup-bun@v2
        with: { bun-version-file: package.json }
      - uses: actions/setup-node@v6
        with: { node-version-file: .node-version }
      - run: bun install --frozen-lockfile
      - run: bun run test:mobile
  bundle:
    needs: changes
    if: "${{ needs.changes.outputs.mobile == 'true' }}"
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v6
      - uses: oven-sh/setup-bun@v2
        with: { bun-version-file: package.json }
      - uses: actions/setup-node@v6
        with: { node-version-file: .node-version }
      - run: bun install --frozen-lockfile
      - run: bun run bundle
      - run: bun run expo:check
  ci-ok:
    needs: [changes, lint, typecheck, api, mobile, bundle]
    if: always()
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - env:
          RESULTS: '${{ toJSON(needs) }}'
        run: 'echo "$RESULTS" | jq -e "all(.[]; .result == \"success\" or .result == \"skipped\")"'
