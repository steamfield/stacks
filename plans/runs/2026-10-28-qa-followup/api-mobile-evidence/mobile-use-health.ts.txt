import { useQuery } from '@tanstack/react-query';
import { api } from '../api';
import { authHeaders, useSessionToken } from '../auth-token';
export function useHealth() {
  const getToken = useSessionToken();
  return useQuery({ queryKey: ['health'], queryFn: async () => {
    const res = await api.health.$get({}, { headers: await authHeaders(getToken) });
    if (!res.ok) throw new Error('request failed'); return res.json();
  } });
}
