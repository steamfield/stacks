import { expect, test } from "@playwright/test";

test("model search changes request; streamed tool and export survive reload", async ({
  page,
  request,
}) => {
  const errors: string[] = [];
  page.on("pageerror", (error) => errors.push(error.message));
  page.on("console", (msg) => {
    if (msg.type() === "error") errors.push(msg.text());
  });
  await page.goto("/");
  await expect(page).toHaveURL(/\/c\/[0-9a-f-]+$/);
  await page.getByRole("combobox", { name: "Model" }).click();
  await page.getByPlaceholder("Search models").fill("Second");
  await page.getByRole("option", { name: /Second/ }).click();
  await request.get("<local-url>/control/tool");
  await page.getByRole("textbox", { name: "Message" }).fill("What time is it?");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByText(/Current time:/)).toBeVisible();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
  const captured = await (
    await request.get("<local-url>/control/requests")
  ).json();
  expect(captured[0].model).toBe("example/second");
  await page.reload();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
  const download = page.waitForEvent("download");
  await page.getByRole("link", { name: "Export" }).click();
  const saved = await download;
  const path = await saved.path();
  const { readFile } = await import("node:fs/promises");
  const body = await readFile(path, "utf8");
  expect(body).toContain("example/second");
  expect(body).toContain("Tool:");
  expect(body).toContain("tokens");
  expect(body).toContain("inputTokens");
  expect(body).toContain("outputTokens");
  expect(body).toContain("servedModel");
  expect(body).toContain("finishReason");
  expect(body).toContain('"cost":0.001');
  expect(errors).toEqual([]);
  await page.getByRole("link", { name: "Extract contact" }).click();
  await page.getByRole("textbox", { name: "Text" }).fill("Ada Lovelace");
  await page.getByRole("button", { name: "Extract" }).click();
  await expect(page.locator("pre")).toContainText("Ada Lovelace");
});
test("stop persists partial reply after reload", async ({ page, request }) => {
  await page.goto("/");
  await request.get("<local-url>/control/hang");
  await page.getByRole("textbox", { name: "Message" }).fill("Pause");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
  const stopped = page.waitForResponse((response) =>
    response.url().endsWith("/stop"),
  );
  await page.getByRole("button", { name: "Stop" }).click();
  expect((await stopped).status()).toBe(204);
  await page.reload();
  await expect(page.getByText(/interrupted/)).toBeVisible();
});

test("copy uses writeText fallback and reports clipboard failure", async ({
  page,
}) => {
  await page.addInitScript(() => {
    const state = window as unknown as Record<string, unknown>;
    state.rejectCopy = false;
    Object.defineProperty(window, "ClipboardItem", {
      value: undefined,
      configurable: true,
    });
    Object.defineProperty(navigator, "clipboard", {
      value: {
        writeText: async (text: string) => {
          if (state.rejectCopy) throw new Error("denied");
          state.copied = text;
        },
      },
      configurable: true,
    });
  });
  await page.goto("/");
  await page.getByRole("textbox", { name: "Message" }).fill("Hello");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
  await page.getByRole("button", { name: "Copy as Markdown" }).click();
  await expect(page.getByText("Copied", { exact: true })).toBeVisible();
  expect(
    await page.evaluate(
      () => (window as unknown as Record<string, unknown>).copied,
    ),
  ).toContain("Hello from the local provider.");
  await page.evaluate(() => {
    (window as unknown as Record<string, unknown>).rejectCopy = true;
  });
  await page.getByRole("button", { name: "Copy as Markdown" }).click();
  await expect(page.getByText("Copy failed", { exact: true })).toBeVisible();
});

test("shows Retry only for a retryable provider error", async ({
  page,
  request,
}) => {
  await page.goto("/");
  await request.get("<local-url>/control/fail?status=401");
  await page.getByRole("textbox", { name: "Message" }).fill("Hello");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByRole("alert")).toContainText("rejected the API key");
  await expect(page.getByRole("button", { name: "Retry" })).toHaveCount(0);
  await request.get("<local-url>/control/fail?status=402");
  await page.getByRole("textbox", { name: "Message" }).fill("Hello again");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByRole("alert")).toContainText("credits");
  await expect(page.getByRole("button", { name: "Retry" })).toBeVisible();
  await page.getByRole("button", { name: "Retry" }).click();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
});

test("falls back to the configured model when discovery is unavailable", async ({
  page,
  request,
}) => {
  await request.get("<local-url>/control/models-down");
  try {
    await page.goto("/");
    await expect(page.getByRole("combobox", { name: "Model" })).toContainText(
      "prism-ml/ternary-bonsai-2-9b",
    );
    await page.getByRole("textbox", { name: "Message" }).fill("Hello");
    await page.getByRole("button", { name: "Send" }).click();
    await expect(
      page.getByText("Hello from the local provider."),
    ).toBeVisible();
  } finally {
    await request.get("<local-url>/control/models-up");
  }
});

test("ClipboardItem receives the export promise and copy is disabled while streaming", async ({
  page,
  request,
}) => {
  await page.addInitScript(() => {
    const state = window as unknown as Record<string, unknown>;
    class FakeItem {
      constructor(public data: Record<string, Promise<Blob>>) {}
    }
    Object.defineProperty(window, "ClipboardItem", {
      value: FakeItem,
      configurable: true,
    });
    Object.defineProperty(navigator, "clipboard", {
      value: {
        write: async (items: FakeItem[]) => {
          state.copied = await (await items[0]?.data["text/plain"])?.text();
        },
      },
      configurable: true,
    });
  });
  await page.goto("/");
  await page.getByRole("textbox", { name: "Message" }).fill("Hello");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(page.getByText("Hello from the local provider.")).toBeVisible();
  await page.getByRole("button", { name: "Copy as Markdown" }).click();
  await expect(page.getByText("Copied", { exact: true })).toBeVisible();
  expect(
    await page.evaluate(
      () => (window as unknown as Record<string, unknown>).copied,
    ),
  ).toContain("Hello from the local provider.");
  await request.get("<local-url>/control/hang");
  await page.getByRole("textbox", { name: "Message" }).fill("Another reply");
  await page.getByRole("button", { name: "Send" }).click();
  await expect(
    page.getByRole("button", { name: "Copy as Markdown" }),
  ).toBeDisabled();
  const stop = page.waitForResponse((response) =>
    response.url().endsWith("/stop"),
  );
  await page.getByRole("button", { name: "Stop" }).click();
  await stop;
});
