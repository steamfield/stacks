import { Effect, Layer, Redacted } from "effect";
import { expect, it } from "vitest";
import { chatProgram } from "../app/.server/chat";
import { AppConfig } from "../app/.server/config";
import { extractProgram } from "../app/.server/extract";
import { Limits } from "../app/.server/services/limits";
import { ChatStore } from "../app/.server/services/store";
import { startStubOpenRouter } from "./helpers/stub-openrouter";

const id = "11111111-1111-4111-8111-111111111111";
const message = {
  id: "22222222-2222-4222-8222-222222222222",
  role: "user",
  parts: [{ type: "text", text: "Hello" }],
};
const post = (body: unknown) =>
  new Request("<local-url>/api/chat", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify(body),
  });
it("refuses missing key without a provider request", async () => {
  const stub = await startStubOpenRouter();
  try {
    const layer = Layer.mergeAll(
      Layer.succeed(AppConfig, { baseUrl: stub.url, model: "example/model" }),
      ChatStore.layerMemory,
      Limits.layer,
    );
    const result = await Effect.runPromise(
      chatProgram(post({ id, message })).pipe(Effect.provide(layer)),
    );
    expect(result.status).toBe(503);
    expect(await result.json()).toEqual({
      message: "Configure an OpenRouter API key.",
      retryable: false,
    });
    const extracted = await Effect.runPromise(
      extractProgram(post({ text: "Ada" })).pipe(Effect.provide(layer)),
    );
    expect(extracted).toBeInstanceOf(Response);
    expect((extracted as Response).status).toBe(503);
    expect(stub.count()).toBe(0);
  } finally {
    await stub.close();
  }
});
it("streams a real provider reply and a tool call against the local stub", async () => {
  const stub = await startStubOpenRouter();
  stub.queueToolCall();
  try {
    const layer = Layer.mergeAll(
      Layer.succeed(AppConfig, {
        key: Redacted.make("dummy"),
        baseUrl: stub.url,
        model: "example/model",
        fallbacks: ["example/fallback"],
        providerSort: "price",
      }),
      ChatStore.layerMemory,
      Limits.layer,
    );
    const response = await Effect.runPromise(
      chatProgram(post({ id, message })).pipe(Effect.provide(layer)),
    );
    const text = await response.text();
    expect(text).toContain("tool-input-available");
    expect(text).toContain("tool-output-available");
    expect(text).toContain("Hello from the local provider.");
    expect(stub.chatRequests).toHaveLength(2);
    expect(stub.chatRequests[0]).toMatchObject({
      models: ["example/model", "example/fallback"],
      provider: {
        sort: "price",
        allow_fallbacks: true,
        data_collection: "allow",
        zdr: false,
      },
    });
    expect(stub.chatRequests[0]).toMatchObject({
      tools: [
        {
          type: "function",
          function: {
            name: "currentTime",
            parameters: {
              type: "object",
              additionalProperties: false,
              required: ["timeZone"],
              properties: { timeZone: { type: "string" } },
            },
          },
        },
      ],
    });
  } finally {
    await stub.close();
  }
});
it("extracts through the real SDK and a local provider", async () => {
  const stub = await startStubOpenRouter();
  try {
    const layer = Layer.mergeAll(
      Layer.succeed(AppConfig, {
        key: Redacted.make("dummy"),
        baseUrl: stub.url,
        model: "example/model",
      }),
      Limits.layer,
    );
    const result = await Effect.runPromise(
      extractProgram(post({ text: "Ada Lovelace" })).pipe(
        Effect.provide(layer),
      ),
    );
    expect(result).toEqual({ name: "Ada Lovelace", email: "<synthetic-email>" });
    expect(stub.chatRequests).toHaveLength(1);
    expect(stub.chatRequests[0]).toMatchObject({
      response_format: {
        type: "json_schema",
        json_schema: {
          schema: {
            type: "object",
            additionalProperties: false,
            required: ["name", "email"],
            properties: { name: { type: "string" }, email: { type: "string" } },
          },
        },
      },
    });
  } finally {
    await stub.close();
  }
});
