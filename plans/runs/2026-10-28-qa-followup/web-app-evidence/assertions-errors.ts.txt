import { Effect, Layer, Redacted } from "effect";
import { expect, it } from "vitest";
import { chatProgram } from "../app/.server/chat";
import { AppConfig } from "../app/.server/config";
import { Limits } from "../app/.server/services/limits";
import { ChatStore } from "../app/.server/services/store";
import { startStubOpenRouter } from "./helpers/stub-openrouter";

for (const [status, retryable] of [
  [401, false],
  [402, true],
  [404, false],
  [429, true],
  [500, true],
] as const)
  it(`maps provider ${status} into safe retryability`, async () => {
    const stub = await startStubOpenRouter();
    stub.failCompletions(status, 2);
    try {
      const layer = Layer.mergeAll(
        Layer.succeed(AppConfig, {
          key: Redacted.make("dummy"),
          baseUrl: stub.url,
          model: "synthetic/local",
        }),
        ChatStore.layerMemory,
        Limits.layer,
      );
      const request = new Request("<local-url>/api/chat", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          id: "11111111-1111-4111-8111-111111111111",
          message: {
            id: "22222222-2222-4222-8222-222222222222",
            role: "user",
            parts: [{ type: "text", text: "Hi" }],
          },
        }),
      });
      const response = await Effect.runPromise(
        chatProgram(request).pipe(Effect.provide(layer)),
      );
      const text = await response.text();
      const frame = text
        .split("\n")
        .filter((line) => line.startsWith("data: "))
        .map((line) => line.slice(6))
        .filter((line) => line !== "[DONE]")
        .map((line) => JSON.parse(line))
        .find((event) => event.type === "error");
      expect(frame).toBeDefined();
      const error = JSON.parse(frame.errorText);
      expect(error.retryable).toBe(retryable);
      expect(error.message).not.toContain("dummy");
      if (status === 402) expect(error.message).toContain("credits");
    } finally {
      await stub.close();
    }
  });
