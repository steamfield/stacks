import { expect, it } from "vitest";
import { chunks, validateRecording } from "../evals/recording";
import { scoreAnswer, scoreContact, validateReport } from "../evals/score";

it("scores results and rejects empty or failing reports", () => {
  expect(scoreAnswer("Hello", "Hello")).toBe("pass");
  expect(scoreAnswer("No", "Hello")).toBe("fail");
  expect(scoreContact({ name: "A", email: "a" })).toBe("pass");
  expect(() => validateReport([])).toThrow();
  const valid = {
    name: "case",
    model: "provider/model",
    verdict: "pass" as const,
    finishReason: "stop",
    latencyMs: 10,
    tokens: 20,
    cost: 0.01,
    provider: "provider",
    servedModel: "provider/model",
  };
  expect(validateReport([valid])).toEqual([valid]);
  expect(() => validateReport([{ ...valid, verdict: "error" }])).toThrow();
  expect(() =>
    validateReport([{ ...valid, cost: undefined } as never]),
  ).toThrow();
});
it("validates synthetic recordings and seven-byte chunks", () => {
  const frame = (delta: unknown, finish_reason: string) =>
    `data: ${JSON.stringify({ model: "synthetic/local", choices: [{ delta, finish_reason }] })}\n\ndata: [DONE]`;
  const samples = [
    frame({ content: "hello" }, "stop"),
    frame({ tool_calls: [{ id: "call" }] }, "tool_calls"),
    frame({ content: "done" }, "stop"),
  ];
  expect(() => validateRecording(samples)).not.toThrow();
  expect(() => validateRecording(samples.slice(0, 1))).toThrow();
  expect(chunks(samples[0] ?? "").join("")).toBe(samples[0]);
  expect(chunks(samples[0] ?? "").every((c) => c.length <= 7)).toBe(true);
});
