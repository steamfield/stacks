import { it } from "@effect/vitest";
import { Effect } from "effect";
import { TestClock } from "effect/testing";
import { expect } from "vitest";
import { Limits } from "../app/.server/services/limits";

it.effect(
  "refills one token without a boundary double burst and releases slots once",
  () =>
    Effect.gen(function* () {
      const limits = yield* Limits;
      for (let i = 0; i < 10; i++) expect(yield* limits.take).toBe(true);
      expect(yield* limits.take).toBe(false);
      yield* TestClock.adjust("6 seconds");
      expect(yield* limits.take).toBe(true);
      expect(yield* limits.take).toBe(false);
      const first = yield* limits.acquire;
      const second = yield* limits.acquire;
      const third = yield* limits.acquire;
      expect(first).toBeTypeOf("function");
      expect(second).toBeTypeOf("function");
      expect(third).toBeTypeOf("function");
      expect(yield* limits.acquire).toBeUndefined();
      first?.();
      first?.();
      const replacement = yield* limits.acquire;
      expect(replacement).toBeTypeOf("function");
      expect(yield* limits.acquire).toBeUndefined();
      replacement?.();
      second?.();
      third?.();
    }).pipe(Effect.provide(Limits.layer)),
);
