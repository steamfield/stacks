import { it } from "@effect/vitest";
import { Effect, Layer } from "effect";
import { TestClock } from "effect/testing";
import { expect } from "vitest";
import { AppConfig } from "../app/.server/config";
import { listModels } from "../app/.server/models";
import { startStubOpenRouter } from "./helpers/stub-openrouter";

it.effect(
  "reads 501 models across pages, caches and refreshes after ten minutes",
  () =>
    Effect.gen(function* () {
      const stub = yield* Effect.promise(() => startStubOpenRouter());
      stub.paginateModels();
      try {
        const layer = Layer.succeed(AppConfig, {
          baseUrl: stub.url,
          model: "provider/last",
        });
        const first = yield* listModels.pipe(Effect.provide(layer));
        expect(first).toHaveLength(501);
        expect(first.at(-1)?.id).toBe("provider/last");
        expect(stub.count()).toBe(2);
        yield* listModels.pipe(Effect.provide(layer));
        expect(stub.count()).toBe(2);
        yield* TestClock.adjust("10 minutes");
        yield* listModels.pipe(Effect.provide(layer));
        expect(stub.count()).toBe(4);
      } finally {
        yield* Effect.promise(() => stub.close());
      }
    }),
);
