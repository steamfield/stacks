import { Effect, Layer, Redacted } from "effect";
import { expect, it } from "vitest";
import { chatProgram } from "../app/.server/chat";
import { AppConfig } from "../app/.server/config";
import { Limits } from "../app/.server/services/limits";
import { ChatStore } from "../app/.server/services/store";
import { startStubOpenRouter } from "./helpers/stub-openrouter";

const frame = (delta: unknown, finish_reason: string | null = null) =>
  `data: ${JSON.stringify({ id: "synthetic", object: "chat.completion.chunk", created: 1, model: "synthetic/local", choices: [{ index: 0, delta, finish_reason }] })}\n\n`;
const synthetic =
  frame({ role: "assistant" }) +
  frame({ content: "Synthetic replay text" }) +
  frame({}, "stop") +
  "data: [DONE]\n\n";
for (const chunked of [false, true])
  it(`replays explicitly synthetic response ${chunked ? "in seven-byte chunks" : "whole"} through real provider and chat program`, async () => {
    const stub = await startStubOpenRouter();
    stub.replay([synthetic], chunked);
    try {
      const layer = Layer.mergeAll(
        Layer.succeed(AppConfig, {
          key: Redacted.make("dummy"),
          baseUrl: stub.url,
          model: "synthetic/local",
        }),
        ChatStore.layerMemory,
        Limits.layer,
      );
      const request = new Request("<local-url>/api/chat", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          id: "11111111-1111-4111-8111-111111111111",
          message: {
            id: "22222222-2222-4222-8222-222222222222",
            role: "user",
            parts: [{ type: "text", text: "Hello" }],
          },
        }),
      });
      const response = await Effect.runPromise(
        chatProgram(request).pipe(Effect.provide(layer)),
      );
      expect(await response.text()).toContain("Synthetic replay text");
      expect(stub.chatRequests).toHaveLength(1);
    } finally {
      await stub.close();
    }
  });

for (const chunked of [false, true])
  it(`replays synthetic two-step tool turn ${chunked ? "chunked" : "whole"}`, async () => {
    const call =
      frame({ role: "assistant" }) +
      frame({
        tool_calls: [
          {
            index: 0,
            id: "synthetic-tool",
            type: "function",
            function: { name: "currentTime", arguments: '{"timeZone":"UTC"}' },
          },
        ],
      }) +
      frame({}, "tool_calls") +
      "data: [DONE]\n\n";
    const stub = await startStubOpenRouter();
    stub.replay([call, synthetic], chunked);
    try {
      const layer = Layer.mergeAll(
        Layer.succeed(AppConfig, {
          key: Redacted.make("dummy"),
          baseUrl: stub.url,
          model: "synthetic/local",
        }),
        ChatStore.layerMemory,
        Limits.layer,
      );
      const request = new Request("<local-url>/api/chat", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          id: "11111111-1111-4111-8111-111111111111",
          message: {
            id: "22222222-2222-4222-8222-222222222222",
            role: "user",
            parts: [{ type: "text", text: "time?" }],
          },
        }),
      });
      const response = await Effect.runPromise(
        chatProgram(request).pipe(Effect.provide(layer)),
      );
      const text = await response.text();
      expect(text).toContain("tool-output-available");
      expect(text).toContain("Synthetic replay text");
      expect(stub.chatRequests).toHaveLength(2);
    } finally {
      await stub.close();
    }
  });
